/** * Infomedics demo — Contact form backend * --------------------------------------- * On each form submission this: * 1. Logs the submission to this Sheet (audit trail / fallback). * 2. Sends an email to info@salesuk.cm.puzzel.com with the enquiry — * this is the direct "trigger an email" requirement, and (if that * address is configured as an inbound email channel in the salesuk * Puzzel Case Management tenant) it should land as a ticket there * the same way a real customer email would. * 3. Also creates a Ticket directly via the Puzzel Case Management API * (tenant: salesuk), so the enquiry shows up as a live case even if * step 2's address isn't wired to auto-create tickets. * * NOTE — step 2 and step 3 both run on every submission. If * info@salesuk.cm.puzzel.com IS a live email channel for this tenant, * you'll get two entries per submission (one from the inbound email, * one from the direct API call). If that turns out to be the case for * a live demo, comment out whichever path you don't want — see * DISABLE_EMAIL_TRIGGER / DISABLE_TICKET_API below. * * ============================================================ * SETUP * ============================================================ * 1. Extensions -> Apps Script on your Google Sheet, delete the * placeholder code, paste this whole file in. * 2. Project Settings (the gear icon, left sidebar) -> Script Properties * -> Add script property, and add these three: * * PUZZEL_TENANT salesuk * PUZZEL_CLIENT_ID * PUZZEL_CLIENT_SECRET * * Do this through the Script Properties UI, not by pasting the values * into this file. This file (and the Sheet it's bound to) may end up * copied into the same public GitHub repo as the demo page — Script * Properties keep the secret out of anything that gets committed. * Since these credentials were shared over chat to build this, it's * worth rotating the Channel Secret Key in Puzzel once testing is done. * 3. Deploy -> New deployment -> type "Web app" -> execute as "Me", * access "Anyone". Copy the /exec URL. * 4. In infomedics-zorgconsument-demo.html, set APPS_SCRIPT_URL to that * URL and re-upload the file. * * Puzzel API reference used: https://demo.cm.puzzel.com/api-docs/public/index.html * (POST /api/v1/oauth_tokens, POST /api/v1/tickets) */ const DISABLE_EMAIL_TRIGGER = false; // set true to stop sending the direct email const DISABLE_TICKET_API = false; // set true to stop calling the Puzzel API directly const PUZZEL_EMAIL_ADDRESS = 'info@salesuk.cm.puzzel.com'; function doPost(e) { const data = JSON.parse(e.postData.contents); logToSheet_(data); if (!DISABLE_EMAIL_TRIGGER) { try { sendPuzzelEmail_(data); updateStatus_('emailStatus', 'sent'); } catch (err) { updateStatus_('emailStatus', 'failed: ' + (err && err.message ? err.message : err)); } } if (!DISABLE_TICKET_API) { try { createPuzzelTicket_(data); updateStatus_('ticketStatus', 'created'); } catch (err) { updateStatus_('ticketStatus', 'failed: ' + (err && err.message ? err.message : err)); } } return ContentService.createTextOutput(JSON.stringify({ status: 'ok' })) .setMimeType(ContentService.MimeType.JSON); } /* ---------------- Sheet logging (audit trail / fallback) ---------------- */ const COLS = { submittedAt: 1, firstName: 2, lastName: 3, email: 4, phone: 5, paymentRef: 6, subject: 7, message: 8, emailStatus: 9, ticketStatus: 10 }; function logToSheet_(data) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow([ 'Submitted at', 'First name', 'Last name', 'Email', 'Phone', 'Payment reference', 'Subject', 'Message', 'Email to Puzzel status', 'Puzzel ticket API status' ]); } const row = [ data.submittedAt || new Date().toISOString(), data.firstName || '', data.lastName || '', data.email || '', "'" + (data.phone || ''), // leading ' forces text (avoids +44 being read as a formula) "'" + (data.paymentRef || ''), // same for long numeric reference codes data.subject || '', data.message || '', '', '' ]; const targetRow = sheet.getLastRow() + 1; sheet.getRange(targetRow, 1, 1, row.length).setValues([row]); sheet.getRange(targetRow, 1, 1, row.length).setNumberFormat('@'); } function updateStatus_(colKey, status) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const lastRow = sheet.getLastRow(); sheet.getRange(lastRow, COLS[colKey]).setValue(status); } /* ---------------- Direct email trigger ---------------- */ function sendPuzzelEmail_(data) { const subject = '[Infomedics demo] ' + (data.subject || 'Website enquiry') + ' — ' + (data.firstName || '') + ' ' + (data.lastName || ''); const body = [ 'New enquiry submitted via the Infomedics demo contact form.', '', 'Name: ' + (data.firstName || '') + ' ' + (data.lastName || ''), 'Email: ' + (data.email || ''), 'Phone: ' + (data.phone || '-'), 'Payment reference: ' + (data.paymentRef || '-'), 'Subject: ' + (data.subject || '-'), '', 'Message:', data.message || '' ].join('\n'); const options = {}; if (data.email) options.replyTo = data.email; MailApp.sendEmail(PUZZEL_EMAIL_ADDRESS, subject, body, options); } /* ---------------- Puzzel Case Management API ---------------- */ function getPuzzelToken_() { const props = PropertiesService.getScriptProperties(); const tenant = props.getProperty('PUZZEL_TENANT'); const clientId = props.getProperty('PUZZEL_CLIENT_ID'); const clientSecret = props.getProperty('PUZZEL_CLIENT_SECRET'); if (!tenant || !clientId || !clientSecret) { throw new Error('Missing PUZZEL_TENANT / PUZZEL_CLIENT_ID / PUZZEL_CLIENT_SECRET script property'); } const res = UrlFetchApp.fetch('https://' + tenant + '.cm.puzzel.com/api/v1/oauth_tokens', { method: 'post', contentType: 'application/json', payload: JSON.stringify({ client_id: clientId, client_secret: clientSecret }), muteHttpExceptions: true }); if (res.getResponseCode() >= 300) { throw new Error('OAuth token request failed (' + res.getResponseCode() + '): ' + res.getContentText()); } const body = JSON.parse(res.getContentText()); if (!body.access_token) throw new Error('No access_token in response: ' + res.getContentText()); return { token: body.access_token, tenant: tenant }; } function createPuzzelTicket_(data) { const auth = getPuzzelToken_(); const bodyText = [ 'New enquiry submitted via the Infomedics demo contact form.', '', 'Name: ' + (data.firstName || '') + ' ' + (data.lastName || ''), 'Email: ' + (data.email || ''), 'Phone: ' + (data.phone || '-'), 'Payment reference: ' + (data.paymentRef || '-'), '', 'Message:', data.message || '' ].join('\n'); const payload = { subject: '[Infomedics demo] ' + (data.subject || 'Website enquiry'), body: bodyText, priority: 'normal', status: 'open', customer: { email: data.email || '', first_name: data.firstName || '', last_name: data.lastName || '' }, tags: ['Puzzel Demo', 'Infomedics Website'] }; if (data.phone) payload.customer.phone_number = data.phone; const res = UrlFetchApp.fetch('https://' + auth.tenant + '.cm.puzzel.com/api/v1/tickets', { method: 'post', contentType: 'application/json', headers: { Authorization: 'Bearer ' + auth.token }, payload: JSON.stringify(payload), muteHttpExceptions: true }); if (res.getResponseCode() >= 300) { throw new Error('Ticket creation failed (' + res.getResponseCode() + '): ' + res.getContentText()); } } /** * Run this once manually from the Apps Script editor (select it in the * function dropdown, click Run) to confirm the token exchange works, * without creating a test ticket. Check the execution log for the result. */ function testTokenOnly() { const auth = getPuzzelToken_(); Logger.log('Got token OK, expires in whatever the API returned. Tenant: ' + auth.tenant); }
